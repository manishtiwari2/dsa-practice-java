import java.io.*;

public class Main{
    public static void main(String[] args) throws Exception{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        
        int w = Integer.parseInt(br.readLine());
        if(w%2 ==0 && w > 2){
            System.out.println("YES");
        }else{
            System.out.println("NO");
        }
    }
}
