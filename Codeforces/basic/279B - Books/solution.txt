import java.util.*;

public class Main {
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        
        int n = sc.nextInt();
        int t = sc.nextInt();
        
        int[] books = new int[n];
        for(int i = 0; i < n; i++){
            books[i] = sc.nextInt();
        }
        
        int left = 0;
        int right = 0;
        int sum = 0;
        int res = 0;
        
        while(right < n){
            sum += books[right];
            
            while(sum > t){
                sum -= books[left];
                left++;
            }
            res = Math.max(res, right - left + 1);
            right++;
        }
        System.out.println(res);
        sc.close();
    }
}
